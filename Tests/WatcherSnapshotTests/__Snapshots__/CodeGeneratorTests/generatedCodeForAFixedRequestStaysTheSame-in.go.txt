package main

import (
	"fmt"
	"io"
	"net/http"
	"strings"
)

func main() {
	req, _ := http.NewRequest("POST", "https://api.example.com/v1/users?page=2&sort=name", strings.NewReader("{\"name\":\"O'Brien\",\"tags\":[\"a\",\"b\"]}"))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer it's-a-token")
	res, err := http.DefaultClient.Do(req)
	if err != nil { panic(err) }
	defer res.Body.Close()
	out, _ := io.ReadAll(res.Body)
	fmt.Println(res.StatusCode, string(out))
}
